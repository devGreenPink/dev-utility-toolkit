# loop_policy.md
max_rounds: 3 ต่อขั้น
on_fail:
  1. อ่าน error ทั้งหมด สรุป root cause ก่อนแก้
  2. สาเหตุเดิมซ้ำ 2 รอบ: เปลี่ยนแนวทาง ไม่ใช่แก้จุดเดิม
  3. ห้ามแก้ test เดิม, fixture, config ของ CI
  4. รอบที่ 3 ยังไม่ผ่าน: หยุด แล้วเขียนรายงาน
     (สิ่งที่ลอง, สมมติฐาน, สิ่งที่ต้องการจากคน)

#!/usr/bin/env bash
# .claude/hooks/protect-files.sh · รับ JSON ของ tool call ทาง stdin
input=$(cat)
path=$(echo "$input" | jq -r '.tool_input.file_path // empty')
for p in ".env" "docs/" ".github/" "tests/test_orders.py"; do
  if [[ "$path" == *"$p"* ]]; then
    echo "BLOCKED: $path เป็นไฟล์ที่ทีมห้ามแก้" >&2
    exit 2
  fi
done
exit 0
// .claude/settings.json
{
  "hooks": {
    "PreToolUse": [{
      "matcher": "Edit|Write",
      "hooks": [{ "type": "command", "command": "bash .claude/hooks/protect-files.sh" }]
    }]
  }
}

# รันใน CI job หรือ shell
claude -p "รัน test ทั้งหมด ถ้ามี test ล้มเหลว วิเคราะห์ root cause และเสนอ patch" \
  --allowedTools "Bash(pytest *),Read,Grep" \
  --permission-mode dontAsk \
  --max-turns 20 \
  --output-format json > result.json
jq -r '.result' result.json

MCP (Model Context Protocol)
MCP คืออะไร มาตรฐานเปิด

มาตรฐานเปิดให้ agent เรียกใช้ tool และข้อมูลจากระบบภายนอก

ใครคุยกับใคร
Claude Code (MCP client) → MCP server → GitHub API / Jira / DB ภายใน
server ให้ 3 อย่าง: tools, resources, prompts · transport: http (remote) หรือ stdio (local)
ชื่อ tool ใน Claude Code คือ mcp__<server>__<tool> ใช้เขียน permission rule และ hook matcher ได้
เพิ่ม server ด้วย claude mcp add 
# remote server แบบ http
claude mcp add --transport http github https://api.githubcopilot.com/mcp/

# local server แบบ stdio (ส่ง env ให้ process)
claude mcp add --transport stdio internal-tools \
  --env API_TOKEN=changeme -- python -m internal_tools.server

# แชร์ให้ทีมผ่าน .mcp.json ใน repo
claude mcp add --transport http --scope project \
  order-stats https://mcp.internal.example/mcp

claude mcp list      # ดูทั้งหมด · get <ชื่อ> ดูรายละเอียด · remove <ชื่อ> ถอด

  "mcpServers": {
    "github": { "type": "http", "url": "https://api.githubcopilot.com/mcp/" },
    "internal-tools": {
      "type": "stdio",
      "command": "python",
      "args": ["-m", "internal_tools.server"],

  }

สร้าง MCP server เอง (Python, FastMCP) 
# internal_tools/server.py · pip install "mcp[cli]" httpx
from mcp.server.fastmcp import FastMCP
import httpx, os

mcp = FastMCP("internal-tools")

@mcp.tool()
def get_order_stats(sku: str, month: str) -> dict:
    """ยอดสั่งซื้อของสินค้าต่อเดือนจากระบบภายใน (อ่านอย่างเดียว)"""
    r = httpx.get(f"{os.environ['STATS_API']}/skus/{sku}",
                  params={"month": month},
                  headers={"Authorization": os.environ["API_TOKEN"]})
    r.raise_for_status()
    return r.json()

if __name__ == "__main__":
    mcp.run()   # stdio transport

Messages API (Python) 
import anthropic

client = anthropic.Anthropic()   # อ่าน ANTHROPIC_API_KEY จาก env
message = client.messages.create(
    model="<model-id>",           # เลือกรุ่นจากเอกสาร
    max_tokens=1024,
    system="คุณคือผู้ตรวจ spec ของทีม ตอบเป็น JSON ตาม schema",
    messages=[{"role": "user",
               "content": f"ตรวจ spec นี้ตาม rubric:\n{rubric}\n\n{spec_text}"}],
)
print(message.content[0].text)
system = Role + Constraint + Format · ขอ JSON ตาม schema แล้ว validate ก่อนใช้ · API key อยู่ใน secret manager ไม่อยู่ในโค้ด
Agent SDK (Python): agent แบบ Claude Code ในโค้ดของคุณ 
# pip install claude-agent-sdk (ต้องมี Claude Code CLI ในเครื่องที่รัน)
import asyncio
from claude_agent_sdk import query, ClaudeAgentOptions

async def main():
    options = ClaudeAgentOptions(
        system_prompt="คุณคือ reviewer ของทีม ตรวจตาม checklist",
        allowed_tools=["Read", "Grep", "Glob", "Bash(git diff *)"],
        permission_mode="default",
        cwd="/srv/repos/order-api",
        max_turns=15,
    )
    async for message in query(prompt="ตรวจ diff ของ branch ปัจจุบัน สรุปเป็น JSON",
                               options=options):
        print(message)

asyncio.run(main())

ชั้น 2: Permission rules และลำดับการประเมิน deny ชนะเสมอ 
// .claude/settings.json
{
  "permissions": {
    "allow": ["Bash(pytest *)", "Bash(git status *)", "Bash(git diff *)",
              "Edit(order_api/**)", "mcp__github__get_pull_request"],
    "ask":   ["Bash(pip install *)", "Bash(git commit *)"],
    "deny":  ["Read(./.env)", "Bash(git push *)", "Bash(rm -rf *)",
              "WebFetch", "mcp__github__merge_pull_request"]
  }
}
1. deny ตรงเมื่อไร block ทันที จาก scope ใดก็ได้
2. ask ถามก่อน
3. allow ทำได้เลย
ไม่ตรงอะไรเลย ใช้พฤติกรรมของ mode
scope: ทีม .claude/settings.json · ส่วนตัว ~/.claude/settings.json · องค์กร managed · โปรเจกต์ allow Bash(git *) แต่ผู้ใช้ deny Bash(git push *) = push ถูกบล็อก · ชื่อ tool เปล่า ๆ ใน deny = ปิด tool นั้นทั้งตัว
ชั้น 3: Hooks

rule ตัดสินจากชื่อ tool กับ pattern ส่วน hook อ่านรายละเอียดของ tool call ได้ทั้งก้อน จึงจับกรณีที่ rule มองไม่เห็น เช่น เขียนทับ test ผ่าน python -c แทน Edit

PreToolUse: กันแก้ไฟล์ต้องห้าม, บังคับ Refs: ใน commit
Stop: รัน test ชุดเดิมก่อนให้จบ ไม่ผ่านให้ทำต่อ (จำกัดรอบด้วย)
hook ของทีมอยู่ใน git · hook ที่ห้ามใครปิดอยู่ใน managed settings
ชั้น 4: Sandbox จำกัดระดับ OS

จำกัด filesystem และ network ของคำสั่ง Bash ที่ Claude รัน · macOS ใช้ Seatbelt · Linux / WSL2 ใช้ bubblewrap · Windows native ยังไม่รองรับ (ใช้ WSL2 หรือ container) · เปิดด้วย /sandbox หรือ settings

// settings.json
{
  "sandbox": {
    "enabled": true,
    "autoAllowBashIfSandboxed": true,
    "failIfUnavailable": true,
    "excludedCommands": ["docker *"],
    "network": { "allowedDomains": ["api.anthropic.com", "registry.npmjs.org"] },
    "filesystem": { "denyRead": ["./.env", "~/.ssh"], "allowWrite": ["./", "/tmp"] }
  }
}
rule ตัดสินว่า "จะรันไหม" sandbox ตัดสินว่า "รันแล้วทำอะไรได้บ้าง" แม้ Claude ถูกหลอกให้รันคำสั่ง ก็ออกนอกกรอบไม่ได้ · autoAllowBashIfSandboxed ลดการถามโดยยังปลอดภัย
